// Q-N-S1-D2 For that month, gross trip revenue by pickup hour and payment type.
// @column month_start_date = [Measures].[q_month_start_date]
// @column pickup_hour = [Pickup Hour].[Pickup Hour]
// @column payment_type_name = [Payment Type].[Payment Type]
// @column revenue = [Measures].[q_revenue]
WITH
SET [q_month] AS
  Head(Order(Exists([Pickup Date].[Month].[Month].Members, {[Pickup Date].[Calendar Year].&[2024]}), [Measures].[Gross Trip Revenue], BDESC), 1)
MEMBER [Measures].[q_month_start_date] AS [q_month].Item(0).Item(0).Name
MEMBER [Measures].[q_revenue] AS ([Measures].[Gross Trip Revenue], [q_month].Item(0).Item(0))
SELECT {[Measures].[q_month_start_date], [Measures].[q_revenue]} ON COLUMNS,
  NonEmpty([Pickup Hour].[Pickup Hour].[Pickup Hour].Members * [Payment Type].[Payment Type].[Payment Type].Members,
           {[Measures].[q_revenue]}) ON ROWS
FROM [NYC]
