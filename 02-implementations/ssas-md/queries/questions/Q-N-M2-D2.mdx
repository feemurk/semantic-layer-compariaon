// Q-N-M2-D2 Fare per mile by pickup hour for Manhattan -> Manhattan trips in 2024.
// @column pickup_hour = [Pickup Hour].[Pickup Hour]
// @column fare_per_mile = [Measures].[Fare per Mile]
SELECT {[Measures].[Fare per Mile]} ON COLUMNS,
  NonEmpty([Pickup Hour].[Pickup Hour].[Pickup Hour].Members, {[Measures].[Trip Count]}) ON ROWS
FROM [NYC]
WHERE ([Pickup Date].[Calendar Year].&[2024], [Pickup Zone].[Borough].&[Manhattan], [Dropoff Zone].[Borough].&[Manhattan])
