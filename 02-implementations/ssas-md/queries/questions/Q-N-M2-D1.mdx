// Q-N-M2-D1 For Manhattan -> Manhattan trips in 2024, the 10 pickup zones with the highest fare per mile among zones with at least 10,000 such trips.
// @column rank_no = [Measures].[q_rank_no]
// @column pu_location_id = [Measures].[q_pu_location_id]
// @column zone = [Pickup Zone].[Zone]
// @column trips = [Measures].[Trip Count]
// @column fare_per_mile = [Measures].[Fare per Mile]
WITH
SET [q_zones] AS
  Head(Order(Order(Filter(Exists([Pickup Zone].[Zone].[Zone].Members, {[Pickup Zone].[Borough].&[Manhattan]}),
                          [Measures].[Trip Count] >= 10000),
                   [Pickup Zone].[Zone].CurrentMember.MemberValue, BASC),
             [Measures].[Fare per Mile], BDESC), 10)
MEMBER [Measures].[q_rank_no] AS Rank([Pickup Zone].[Zone].CurrentMember, [q_zones])
MEMBER [Measures].[q_pu_location_id] AS [Pickup Zone].[Zone].CurrentMember.Properties("Location ID")
SELECT {[Measures].[q_rank_no], [Measures].[q_pu_location_id], [Measures].[Trip Count],
        [Measures].[Fare per Mile]} ON COLUMNS,
  [q_zones] ON ROWS
FROM [NYC]
WHERE ([Pickup Date].[Calendar Year].&[2024], [Dropoff Zone].[Borough].&[Manhattan])
