// Q-N-S1-D1 For the 2024 month with the highest revenue, the 15 Manhattan pickup zones with the most revenue.
// @column month_start_date = [Measures].[q_month_start_date]
// @column rank_no = [Measures].[q_rank_no]
// @column pu_location_id = [Measures].[q_pu_location_id]
// @column zone = [Pickup Zone].[Zone]
// @column revenue = [Measures].[q_revenue]
WITH
SET [q_month] AS
  Head(Order(Exists([Pickup Date].[Month].[Month].Members, {[Pickup Date].[Calendar Year].&[2024]}), [Measures].[Gross Trip Revenue], BDESC), 1)
SET [q_zones] AS
  Head(Order(Order(Exists([Pickup Zone].[Zone].[Zone].Members, {[Pickup Zone].[Borough].&[Manhattan]}),
                   [Pickup Zone].[Zone].CurrentMember.MemberValue, BASC),
             ([Measures].[Gross Trip Revenue], [q_month].Item(0).Item(0)), BDESC), 15)
MEMBER [Measures].[q_month_start_date] AS [q_month].Item(0).Item(0).Name
MEMBER [Measures].[q_rank_no] AS Rank([Pickup Zone].[Zone].CurrentMember, [q_zones])
MEMBER [Measures].[q_pu_location_id] AS [Pickup Zone].[Zone].CurrentMember.Properties("Location ID")
MEMBER [Measures].[q_revenue] AS ([Measures].[Gross Trip Revenue], [q_month].Item(0).Item(0))
SELECT {[Measures].[q_month_start_date], [Measures].[q_rank_no], [Measures].[q_pu_location_id],
        [Measures].[q_revenue]} ON COLUMNS,
  [q_zones] ON ROWS
FROM [NYC]
