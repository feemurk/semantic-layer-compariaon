// Q-N-S1-D4 Which 10 dropoff zones received the most 2024 revenue from trips picked up at JFK Airport?
// @column rank_no = [Measures].[q_rank_no]
// @column do_location_id = [Measures].[q_do_location_id]
// @column dropoff_borough = [Measures].[q_dropoff_borough]
// @column dropoff_zone = [Dropoff Zone].[Zone]
// @column revenue = [Measures].[Gross Trip Revenue]
WITH
SET [q_zones] AS
  Head(Order(Order(NonEmpty([Dropoff Zone].[Zone].[Zone].Members, {[Measures].[Gross Trip Revenue]}),
                   [Dropoff Zone].[Zone].CurrentMember.MemberValue, BASC),
             [Measures].[Gross Trip Revenue], BDESC), 10)
MEMBER [Measures].[q_rank_no] AS Rank([Dropoff Zone].[Zone].CurrentMember, [q_zones])
MEMBER [Measures].[q_do_location_id] AS [Dropoff Zone].[Zone].CurrentMember.Properties("Location ID")
MEMBER [Measures].[q_dropoff_borough] AS [Dropoff Zone].[Zone].CurrentMember.Properties("Borough")
SELECT {[Measures].[q_rank_no], [Measures].[q_do_location_id], [Measures].[q_dropoff_borough],
        [Measures].[Gross Trip Revenue]} ON COLUMNS,
  [q_zones] ON ROWS
FROM [NYC]
WHERE ([Pickup Date].[Calendar Year].&[2024], [Pickup Zone].[Zone].[JFK Airport])
